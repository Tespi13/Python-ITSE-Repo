from random import randrange


def get_user_weapon():
    print("SELECT YOUR WEAPON (1-3)")
    print("1. Rock")
    print("2. Paper")
    print("3. Scissors")
    return int(input("Enter your weapon: "))


def get_opponent_weapon():
    return randrange(1, 4)


def determine_winner(user_weapon, opponent_weapon):
    if user_weapon == opponent_weapon:
        print("It's a tie!")
    elif (
        (user_weapon == 1 and opponent_weapon == 3)
        or (user_weapon == 2 and opponent_weapon == 1)
        or (user_weapon == 3 and opponent_weapon == 2)
    ):
        if user_weapon == 1:
            print("You win, rock crushes scissors!")
        elif user_weapon == 2:
            print("You win, paper covers rock!")
        else:
            print("You win, scissors cuts paper!")
    else:
        if opponent_weapon == 1:
            print("You lose, rock crushes scissors!")
        elif opponent_weapon == 2:
            print("You lose, paper covers rock!")
        else:
            print("You lose, scissors cuts paper!")


def main():
    play_again = "y"

    while play_again.lower() == "y":
        user_weapon = get_user_weapon()
        opponent_weapon = get_opponent_weapon()
        determine_winner(user_weapon, opponent_weapon)

        play_again = input("Want to play again (y/n): ")

    print("Completed by, Tony Espinoza")


if __name__ == "__main__":
    main()
